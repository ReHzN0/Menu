// Bu fonksiyon şifreyi gizli bir koda çevirir async function hashPassword(string) { const utf8 = new TextEncoder().encode(string); const hashBuffer = await crypto.subtle.digest('SHA-256', utf8); const hashArray = Array.from(new Uint8Array(hashBuffer)); return hashArray.map(b => b.toString(16).padStart(2, '0')).join(''); } async function secureToggle() { if (!isAdmin) { const entry = prompt("Yönetici şifresini girin:"); if (!entry) return; const entryHash = await hashPassword(entry); // ÖRNEK: '1234' şifresinin SHA-256 karşılığı aşağıdakidir. // Koda bakan kişi sadece bunu görür: const masterHash = "03ac674216f3e15c742060a120301d2326502992999897335a83cfee8d41a1f0"; if (entryHash === masterHash) { isAdmin = true; document.getElementById('mainBody').className = "admin-mode"; document.getElementById('modeText').innerText = "Yönetici Paneli"; initMenu(); } else { alert("Hatalı Şifre!"); } } else { isAdmin = false; document.getElementById('mainBody').className = "viewer-mode"; document.getElementById('modeText').innerText = "Müşteri Menüsü"; initMenu(); } }
